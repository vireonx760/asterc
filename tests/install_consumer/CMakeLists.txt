cmake_minimum_required(VERSION 3.20)
project(AsterASMConsumer LANGUAGES C)
enable_testing()
find_package(AsterASM CONFIG REQUIRED)
foreach(kind backend shared)
    add_executable(consumer_${kind} main.c)
    target_link_libraries(consumer_${kind} PRIVATE AsterASM::${kind})
    add_test(NAME consumer_${kind} COMMAND consumer_${kind})
endforeach()
if(TARGET AsterASM::object)
    foreach(kind object object_static)
        add_executable(consumer_${kind} object.c)
        target_link_libraries(consumer_${kind} PRIVATE AsterASM::${kind})
        add_test(NAME consumer_${kind} COMMAND consumer_${kind})
    endforeach()
endif()
